124void GameWidget::updatePos(){
125 if(gameWon) { update(); return; }
126
127 velX += accX;
128 velY += accY;
129 velX *= (1.0 - friction);
130 velY *= (1.0 - friction);
131 ballX += velX;
132 ballY += velY;
133
134 if(ballX < 0.0){ballX = 0.0; velX = 0.0;}
135 if(ballX > 1.0){ballX = 1.0; velX = 0.0;}
136 if(ballY < 0.0){ballY = 0.0; velY = 0.0;}
137 if(ballY > 1.0){ballY = 1.0; velY = 0.0;}
138
139 for(int i = 0; i < 3; ++i){
140
141 for(const QRectF &wall : walls){
142 double closestX = qBound(wall.left(), ballX, wall.right());
143 double closestY = qBound(wall.top(), ballY, wall.bottom());
144
145 double distX = ballX - closestX;
146 double distY = ballY - closestY;
147 double dist = std::sqrt(distX * distX + distY * distY);
148
149 if(dist <= ballRadius && dist > 0){
150 double nx = distX / dist;
151 double ny = distY / dist;
152
153 ballX = closestX + nx * ballRadius;
154 ballY = closestY + ny * ballRadius;
155
156 double dotProduct = velX * nx + velY * ny;
157 if(dotProduct < 0) {
158 velX -= dotProduct * nx;
159 velY -= dotProduct * ny;
160 }
161 }
162
163 double dx = ballX - goalX;
164 double dy = ballY - goalY;
165 if(std::sqrt(dx*dx + dy*dy) < ballRadius) {
166 gameWon = true;
167 velX = velY = accX = accY = 0.0;
168 }
169 }
170 }
171
172 pulsePhase += 0.05;
173 if(pulsePhase > 2 * M_PI) pulsePhase -= 2 * M_PI;
174
175
176 update();
177}
